Every data value goes through partial "tex.html" (or "tex_url.html" for links).
Go templates start an action at "{{", so a LaTeX group that opens right before a
value is written "{ {{- ... }}", the "-" trimming the space in between.
*/ -}}

% \cvsection{title}: starts a new page instead if fewer than ~4 lines are left,
% so a heading is never stranded at the bottom of a page.
\newcommand{\cvsection}[1]{\par\vspace{0pt plus 4\baselineskip}\penalty-200\vspace{0pt plus -4\baselineskip}
  \bigskip{\large\color{accent}\MakeUppercase{#1}}\par\nopagebreak\medskip}

% \cventry{date}{body}: muted date column, thin rule, then the body.
\newlength{\cvdatewidth}
\setlength{\cvdatewidth}{2.8cm}
\newcommand{\cventry}[2]{\par\noindent
  \parbox[t]{\cvdatewidth}{\raggedright\small\leavevmode\color{muted}#1}%
  \hspace{6pt}{\color{rulegray}\vrule width 0.4pt}\hspace{6pt}%
  \parbox[t]{\dimexpr\linewidth-\cvdatewidth-12.4pt}{\raggedright#2}\par\medskip}

% \cvdetail{text}: the smaller, muted line(s) below an entry's first line.
\newcommand{\cvdetail}[1]{\\{\small\color{muted}#1}}

% \cvitem{text}{year}: one line, year flushed right.
\newcommand{\cvitem}[2]{\par\noindent#1\hfill{\color{muted}#2}\par}

% \cvpub{title}{venue}{authors}
\newcommand{\cvpub}[3]{\par\noindent
  \parbox[t]{\dimexpr\linewidth-3.2cm}{\raggedright#1}\hfill
  \parbox[t]{3cm}{\raggedleft\small\leavevmode\color{accent}#2}\par
  {\small\color{muted}#3}\par\medskip}
